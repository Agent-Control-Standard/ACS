# Test dependencies shared across all reference adapters.
# Production adapter runtime needs only stdlib + rfc8785 (for full JCS
# canonicalization per RFC 8785). Tests additionally need jsonschema
# for canonical spec validation.
#
# EXACT pins, deliberately (PR #22 review: the CI header said "pinned
# dependencies" while four of six were ranges). Ranges let a future
# release shift validator or canonicalization behavior under the suite
# without test signal; bumping a pin is a reviewed diff.
jsonschema==4.25.1
rfc8785==0.1.4
# jsonschema's date-time / uri / hostname / email format checkers
# delegate to optional companion packages. Without these pinned, the
# checkers silently no-op and tests like test_timestamp_is_iso8601
# false-pass (the invalid value goes through, an "expected to fail"
# assertion sees an empty error list, and CI shows green on a real
# wire-format bug). Pinning rfc3339-validator + the format extras
# turns the checker back on so the suite catches the regression.
rfc3339-validator==0.1.4
jsonschema[format]==4.25.1
